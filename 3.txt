## mail/models.py

from django.db import models
from django.contrib.auth.models import User


class Mail(models.Model):

    sender = models.EmailField()

    recipient = models.EmailField()

    subject = models.CharField(max_length=255)

    body = models.TextField()

    timestamp = models.DateTimeField(auto_now_add=True)

    owner = models.ForeignKey(User, on_delete=models.CASCADE)

    def __str__(self):
        return f"{self.subject} ({self.sender} → {self.recipient})"

## mail/views.py

from django.shortcuts import render, redirect
from django.contrib.auth.decorators import login_required
from django.contrib.auth import login, logout
from django.contrib.auth.forms import AuthenticationForm
from django.core.mail import send_mail
from django.conf import settings

from .forms import MailForm, RegisterForm
from .models import Mail


# Inbox View
@login_required
def inbox_view(request):

    mails = Mail.objects.filter(owner=request.user).order_by('-timestamp')

    return render(request, "mail/inbox.html", {"mails": mails})


# Compose Mail View
@login_required
def compose_view(request):

    if request.method == "POST":

        form = MailForm(request.POST)

        if form.is_valid():

            recipient = form.cleaned_data['recipient']

            subject = form.cleaned_data['subject']

            body = form.cleaned_data['body']

            send_mail(subject, body, settings.EMAIL_HOST_USER, [recipient])

            Mail.objects.create(
                sender=settings.EMAIL_HOST_USER,
                recipient=recipient,
                subject=subject,
                body=body,
                owner=request.user
            )

            return redirect("inbox")

    else:

        form = MailForm()

    return render(request, "mail/compose.html", {"form": form})


# Login View
def login_view(request):

    if request.method == "POST":

        form = AuthenticationForm(request, data=request.POST)

        if form.is_valid():

            user = form.get_user()

            login(request, user)

            return redirect("inbox")

    else:

        form = AuthenticationForm()

    return render(request, "mail/login.html", {"form": form})


# Logout View
@login_required
def logout_view(request):

    logout(request)

    return redirect("login")


# Register View
def register_view(request):

    if request.method == "POST":

        form = RegisterForm(request.POST)

        if form.is_valid():

            user = form.save()

            login(request, user)

            return redirect("inbox")

    else:

        form = RegisterForm()

    return render(request, "mail/register.html", {"form": form})

// Wait until the DOM is fully loaded

document.addEventListener("DOMContentLoaded", function() {

    // Confirmation before sending mail

    const composeForm = document.querySelector("form");

    if (composeForm && window.location.pathname.includes("compose")) {

        composeForm.addEventListener("submit", function(event) {

            const confirmSend = confirm("Are you sure you want to send this mail?");

            if (!confirmSend) {

                event.preventDefault();

            } else {

                showToast("Mail sent successfully!");

            }

        });

    }


    // Highlight inbox items on hover

    const inboxItems = document.querySelectorAll("ul li");

    inboxItems.forEach(item => {

        item.addEventListener("mouseover", () => {

            item.style.backgroundColor = "#e9f5ff";

        });

        item.addEventListener("mouseout", () => {

            item.style.backgroundColor = "#fff";

        });

    });


    // Auto-focus first input field

    const firstInput = document.querySelector("form input");

    if (firstInput) {

        firstInput.focus();

    }


    // Password visibility toggle

    const passwordFields = document.querySelectorAll("input[type='password']");

    passwordFields.forEach(field => {

        const toggle = document.createElement("span");

        toggle.textContent = "Show/Hide";

        toggle.style.cursor = "pointer";

        toggle.style.marginLeft = "10px";

        field.parentNode.appendChild(toggle);

        toggle.addEventListener("click", () => {

            field.type = field.type === "password" ? "text" : "password";

        });

    });


    // Toast notification function

    function showToast(message) {

        const toast = document.createElement("div");

        toast.textContent = message;

        toast.style.position = "fixed";

        toast.style.bottom = "20px";

        toast.style.right = "20px";

        toast.style.background = "#28a745";

        toast.style.color = "#fff";

        toast.style.padding = "10px 20px";

        toast.style.borderRadius = "5px";

        toast.style.boxShadow = "0px 2px 6px rgba(0,0,0,0.2)";

        document.body.appendChild(toast);

        setTimeout(() => toast.remove(), 3000);

    }

});
